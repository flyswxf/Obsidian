## DSU
```cpp
struct DSU {
    vector<int> parent;
    DSU(int n) : parent(n + 1) {
        for (int i = 0; i <= n; i++) parent[i] = i;
    }
    int find(int x) {
        return parent[x] == x ? x : parent[x] = find(parent[x]);
    }
    bool unite(int a, int b) {
        a = find(a);
        b = find(b);
        if (a == b) return false; // 已连通，再加这条边会成环
        parent[a] = b;
        return true;
    }
};
```
## pow
```cpp
long long qpow(long long a, long long b, long long p) {
    long long res = 1 % p;
    a %= p;
    while (b) {
        if (b & 1) res = res * a % p;
        a = a * a % p;
        b >>= 1;
    }
    return res;
}
```

## 线段树
```cpp
#include <iostream>
#include <vector>
using namespace std;
typedef long long ll;
class SegmentTree {
private:
    vector<ll> tree; // 存储线段树节点值（例如区间和）
    vector<ll> lazy; // 懒标记数组
    int n;

    // 用子节点的值更新父节点 (Push Up)
    void pushUp(int node) {
        tree[node] = tree[node * 2] + tree[node * 2 + 1];
    }
    // 下传懒标记 (Push Down)
    void pushDown(int node, int left, int right) {
        if (lazy[node] != 0) {
            int mid = left + (right - left) / 2;
            
            // 更新左子节点：值增加 lazy[node] * 左区间长度
            tree[node * 2] += lazy[node] * (mid - left + 1);
            lazy[node * 2] += lazy[node]; // 累加懒标记
            
            // 更新右子节点：值增加 lazy[node] * 右区间长度
            tree[node * 2 + 1] += lazy[node] * (right - mid);
            lazy[node * 2 + 1] += lazy[node]; // 累加懒标记
            
            // 清空当前节点的懒标记
            lazy[node] = 0;
        }
    }
    // 递归建树
    void build(const vector<int>& arr, int node, int left, int right) {
        if (left == right) {
            tree[node] = arr[left];
            return;
        }
        int mid = left + (right - left) / 2;
        build(arr, node * 2, left, mid);       // 左子树
        build(arr, node * 2 + 1, mid + 1, right); // 右子树
        pushUp(node);
    }
    // 区间更新
    void update(int node, int left, int right, int ql, int qr, ll val) {
        // 如果当前区间被完全包含在查询区间内，直接更新并打上懒标记
        if (ql <= left && right <= qr) {
            tree[node] += val * (right - left + 1);
            lazy[node] += val;
            return;
        }
        
        // 如果没有完全包含，必须先下传懒标记
        pushDown(node, left, right);
        
        int mid = left + (right - left) / 2;
        if (ql <= mid) update(node * 2, left, mid, ql, qr, val);
        if (qr > mid) update(node * 2 + 1, mid + 1, right, ql, qr, val);
        
        pushUp(node); // 更新完子节点后，更新当前节点
    }
    // 区间查询
    ll query(int node, int left, int right, int ql, int qr) {
        // 如果当前区间被完全包含，直接返回该节点的值
        if (ql <= left && right <= qr) {
            return tree[node];
        }
        
        // 查询前必须下传懒标记
        pushDown(node, left, right);
        
        int mid = left + (right - left) / 2;
        ll sum = 0;
        if (ql <= mid) sum += query(node * 2, left, mid, ql, qr);
        if (qr > mid) sum += query(node * 2 + 1, mid + 1, right, ql, qr);
        
        return sum;
    }
public:
    // 构造函数
    SegmentTree(const vector<int>& arr) {
        n = arr.size();
        // 线段树的空间一般开到原数组大小的 4 倍
        tree.assign(4 * n, 0);
        lazy.assign(4 * n, 0);
        if (n > 0) {
            build(arr, 1, 0, n - 1);
        }
    }
    // 外部接口：将区间 [ql, qr] 内的所有元素加上 val
    void update(int ql, int qr, ll val) {
        update(1, 0, n - 1, ql, qr, val);
    }
    // 外部接口：查询区间 [ql, qr] 的和
    ll query(int ql, int qr) {
        return query(1, 0, n - 1, ql, qr);
    }
};

int main() {
    vector<int> arr = {1, 2, 3, 4, 5};
    SegmentTree st(arr);
    // 将区间 [0, 2] 的每个元素加上 2 (数组变为 3, 4, 5, 4, 5)
    st.update(0, 2, 2);
    // 更新后 [0, 2] 的和：3 + 4 + 5 = 12
    cout << "Sum of [0, 2] after update: " << st.query(0, 2) << endl;
}
```
